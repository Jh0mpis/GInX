52template <int INTERPOLATION_ORDER, int DERIVATIVE_ORDER>
53AMREX_GPU_HOST_DEVICE CCTK_ATTRIBUTE_ALWAYS_INLINE inline void
54scalar_barycentric_derivative(amrex::GpuArray<CCTK_REAL, 3> &df,
55 amrex::Array4<CCTK_REAL const> const &f,
56 int const &i0, int const &j0, int const &k0,
57 CCTK_REAL const &x, CCTK_REAL const &y,
58 CCTK_REAL const &z,
59 const amrex::GpuArray<double, 3> &dx,
60 const amrex::GpuArray<double, 3> &plo) {
61 // Computing the f(x-dx, y, z) value
62 const CCTK_REAL m1_x = barycentric_cubic_3d<INTERPOLATION_ORDER>(
63 f, i0 - 1, j0, k0, x - dx[0], y, z, dx, plo);
64 // Computing the f(x+dx, y, z) value
65 const CCTK_REAL p1_x = barycentric_cubic_3d<INTERPOLATION_ORDER>(
66 f, i0 + 1, j0, k0, x + dx[0], y, z, dx, plo);
67 // Computing the f(x, y-dy, z) value
68 const CCTK_REAL m1_y = barycentric_cubic_3d<INTERPOLATION_ORDER>(
69 f, i0, j0 - 1, k0, x, y - dx[1], z, dx, plo);
70 // Computing the f(x, y+dy, z) value
71 const CCTK_REAL p1_y = barycentric_cubic_3d<INTERPOLATION_ORDER>(
72 f, i0, j0 + 1, k0, x, y + dx[1], z, dx, plo);
73 // Computing the f(x, y, z-dz) value
74 const CCTK_REAL m1_z = barycentric_cubic_3d<INTERPOLATION_ORDER>(
75 f, i0, j0, k0 - 1, x, y, z - dx[2], dx, plo);
76 // Computing the f(x, y, z+dz) value
77 const CCTK_REAL p1_z = barycentric_cubic_3d<INTERPOLATION_ORDER>(
78 f, i0, j0, k0 + 1, x, y, z + dx[2], dx, plo);
79
80 // fill the gradient
81 df[0] = (p1_x - m1_x) / (2.0 * dx[0]);
82 df[1] = (p1_y - m1_y) / (2.0 * dx[1]);
83 df[2] = (p1_z - m1_z) / (2.0 * dx[2]);
84} // scalar_barycentric_derivative
