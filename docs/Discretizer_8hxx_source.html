57template <int DERIVATIVE_ORDER>
58AMREX_GPU_HOST_DEVICE CCTK_ATTRIBUTE_ALWAYS_INLINE inline void
59scalar_barycentric_derivative(amrex::GpuArray<CCTK_REAL, 3> &df,
60 amrex::Array4<CCTK_REAL const> const &f,
61 int const &i0, int const &j0, int const &k0,
62 CCTK_REAL const &x, CCTK_REAL const &y,
63 CCTK_REAL const &z,
64 const amrex::GpuArray<double, 3> &dx,
65 const amrex::GpuArray<double, 3> &plo) {
66 // Computing the f(x-dx, y, z) value
67 const int ip_half_0 = amrex::Math::floor((x + 0.5 * dx[0] - plo[0]) / dx[0]);
68 const int jp_half_0 = amrex::Math::floor((y + 0.5 * dx[1] - plo[1]) / dx[1]);
69 const int kp_half_0 = amrex::Math::floor((z + 0.5 * dx[2] - plo[2]) / dx[2]);
70 const int im_half_0 = amrex::Math::floor((x - 0.5 * dx[0] - plo[0]) / dx[0]);
71 const int jm_half_0 = amrex::Math::floor((y - 0.5 * dx[1] - plo[1]) / dx[1]);
72 const int km_half_0 = amrex::Math::floor((z - 0.5 * dx[2] - plo[2]) / dx[2]);
73
74 const CCTK_REAL p_half_x = barycentric_cubic_3d<DERIVATIVE_ORDER>(
75 f, ip_half_0, j0, k0, x + 0.5 * dx[0], y, z, dx, plo);
76 const CCTK_REAL m_half_x = barycentric_cubic_3d<DERIVATIVE_ORDER>(
77 f, im_half_0, j0, k0, x - 0.5 * dx[0], y, z, dx, plo);
78 // Computing the f(x, y-dy, z) value
79 const CCTK_REAL p_half_y = barycentric_cubic_3d<DERIVATIVE_ORDER>(
80 f, i0, jp_half_0, k0, x, y + 0.5 * dx[1], z, dx, plo);
81 const CCTK_REAL m_half_y = barycentric_cubic_3d<DERIVATIVE_ORDER>(
82 f, i0, jm_half_0, k0, x, y - 0.5 * dx[1], z, dx, plo);
83 // Computing the f(x, y, z-dz) value
84 const CCTK_REAL p_half_z = barycentric_cubic_3d<DERIVATIVE_ORDER>(
85 f, i0, j0, kp_half_0, x, y, z + 0.5 * dx[2], dx, plo);
86 const CCTK_REAL m_half_z = barycentric_cubic_3d<DERIVATIVE_ORDER>(
87 f, i0, j0, km_half_0, x, y, z - 0.5 * dx[2], dx, plo);
88
89 // fill the gradient
90 df[0] = (p_half_x - m_half_x) / (dx[0]);
91 df[1] = (p_half_y - m_half_y) / (dx[1]);
92 df[2] = (p_half_z - m_half_z) / (dx[2]);
93} // scalar_barycentric_derivative
