137template <typename StructType>
138amrex::GpuArray<CCTK_REAL, StructType::n_attributes + 3>
139PhotonsContainer<StructType>::compute_rhs(
140 const amrex::GpuArray<CCTK_REAL, StructType::n_attributes + 3> &u,
141 const CCTK_REAL &t, amrex::Array4<CCTK_REAL const> const &lapse,
142 amrex::Array4<CCTK_REAL const> const &shift,
143 amrex::Array4<CCTK_REAL const> const &metric,
144 amrex::Array4<CCTK_REAL const> const &curv, const CCTK_REAL dt,
145 const amrex::GpuArray<double, 3> &dx, const int lev,
146 const amrex::GpuArray<double, 3> &plo) {
147
148 amrex::GpuArray<CCTK_REAL, StructType::n_attributes + 3> rhs = {
149 0., 0., 0., 0., 0., 0., 0.};
150
151 // Compute the cell's index where the particle is at position (u[0], u[1],
152 // u[2])
153 const int i0 = amrex::Math::floor((u[0] - plo[0]) / dx[0]);
154 const int j0 = amrex::Math::floor((u[1] - plo[1]) / dx[1]);
155 const int k0 = amrex::Math::floor((u[2] - plo[2]) / dx[2]);
156
157 // Interpolate lapse
158 const CCTK_REAL lapse_x =
159 barycentric_cubic_3d<3>(lapse, i0, j0, k0, u[0], u[1], u[2], dx, plo);
160
161 // Interpolate shift
162 const amrex::GpuArray<CCTK_REAL, 3> shift_x = {
163 barycentric_cubic_3d<3>(shift, i0, j0, k0, u[0], u[1], u[2], dx, plo, 0),
164 barycentric_cubic_3d<3>(shift, i0, j0, k0, u[0], u[1], u[2], dx, plo, 1),
165 barycentric_cubic_3d<3>(shift, i0, j0, k0, u[0], u[1], u[2], dx, plo, 2)};
166
167 // Interpolate metric
168 const amrex::GpuArray<CCTK_REAL, 6> gamma_x = {
169 barycentric_cubic_3d<3>(metric, i0, j0, k0, u[0], u[1], u[2], dx, plo,
170 0), // g_11
171 barycentric_cubic_3d<3>(metric, i0, j0, k0, u[0], u[1], u[2], dx, plo,
172 1), // g_12 & g_21
173 barycentric_cubic_3d<3>(metric, i0, j0, k0, u[0], u[1], u[2], dx, plo,
174 2), // g_13 & g_31
175 barycentric_cubic_3d<3>(metric, i0, j0, k0, u[0], u[1], u[2], dx, plo,
176 3), // g_22
177 barycentric_cubic_3d<3>(metric, i0, j0, k0, u[0], u[1], u[2], dx, plo,
178 4), // g_23, g_32
179 barycentric_cubic_3d<3>(metric, i0, j0, k0, u[0], u[1], u[2], dx, plo,
180 5)}; // g_33
181
182 // Interpolate Curvature
183 const amrex::GpuArray<CCTK_REAL, 6> curv_x = {
184 barycentric_cubic_3d<3>(curv, i0, j0, k0, u[0], u[1], u[2], dx, plo,
185 0), // K_11
186 barycentric_cubic_3d<3>(curv, i0, j0, k0, u[0], u[1], u[2], dx, plo,
187 1), // K_12 & g_21
188 barycentric_cubic_3d<3>(curv, i0, j0, k0, u[0], u[1], u[2], dx, plo,
189 2), // K_13 & K_31
190 barycentric_cubic_3d<3>(curv, i0, j0, k0, u[0], u[1], u[2], dx, plo,
191 3), // K_22
192 barycentric_cubic_3d<3>(curv, i0, j0, k0, u[0], u[1], u[2], dx, plo,
193 4), // K_23, K_32
194 barycentric_cubic_3d<3>(curv, i0, j0, k0, u[0], u[1], u[2], dx, plo,
195 5)}; // K_33
196
197 // Interpolate partial lapse
198 amrex::GpuArray<CCTK_REAL, 3> d_lapse_x; //= {0, 0, 0};
199 scalar_barycentric_derivative<3, 2>(d_lapse_x, lapse, i0, j0, k0, u[0], u[1],
200 u[2], dx, plo);
201
202 // Interpolate partial shift
203 amrex::GpuArray<amrex::GpuArray<CCTK_REAL, 3>, 3> d_shift_x;
204 vector_barycentric_derivative<3, 2>(d_shift_x, shift, i0, j0, k0, u[0], u[1],
205 u[2], dx, plo);
206
207 // Interpolate partial metric
208 amrex::GpuArray<amrex::GpuArray<CCTK_REAL, 6>, 3> d_gamma_x;
209 smatrix_barycentric_derivative<3, 2>(d_gamma_x, metric, i0, j0, k0, u[0],
210 u[1], u[2], dx, plo);
211
212 // Compute the inverse of the metric.
213 const CCTK_REAL inv_det_gamma =
214 1.0 / (gamma_x[0] * gamma_x[3] * gamma_x[5] +
215 2. * gamma_x[1] * gamma_x[2] * gamma_x[4] -
216 gamma_x[2] * gamma_x[2] * gamma_x[3] -
217 gamma_x[4] * gamma_x[4] * gamma_x[0] -
218 gamma_x[1] * gamma_x[1] * gamma_x[5]);
219
220 const amrex::GpuArray<CCTK_REAL, 6> gamma_inv_x = {
221 (gamma_x[3] * gamma_x[5] - gamma_x[4] * gamma_x[4]) * inv_det_gamma,
222 (gamma_x[4] * gamma_x[2] - gamma_x[1] * gamma_x[5]) * inv_det_gamma,
223 (gamma_x[1] * gamma_x[4] - gamma_x[2] * gamma_x[3]) * inv_det_gamma,
224 (gamma_x[0] * gamma_x[5] - gamma_x[2] * gamma_x[2]) * inv_det_gamma,
225 (gamma_x[2] * gamma_x[1] - gamma_x[0] * gamma_x[4]) * inv_det_gamma,
226 (gamma_x[0] * gamma_x[3] - gamma_x[1] * gamma_x[1]) * inv_det_gamma};
227
228 const CCTK_REAL &V1 = u[3];
229 const CCTK_REAL &V2 = u[4];
230 const CCTK_REAL &V3 = u[5];
231 const amrex::GpuArray<CCTK_REAL, 3> V_down = {V1, V2, V3};
232
233 // Compute the upper index velocity terms.
234 const amrex::GpuArray<CCTK_REAL, 3> V_up = {
235 gamma_inv_x[0] * V1 + gamma_inv_x[1] * V2 + gamma_inv_x[2] * V3,
236 gamma_inv_x[1] * V1 + gamma_inv_x[3] * V2 + gamma_inv_x[4] * V3,
237 gamma_inv_x[2] * V1 + gamma_inv_x[4] * V2 + gamma_inv_x[5] * V3};
238
239 // Compute the rhs for position
240 rhs[0] = lapse_x * V_up[0] - shift_x[0];
241 rhs[1] = lapse_x * V_up[1] - shift_x[1];
242 rhs[2] = lapse_x * V_up[2] - shift_x[2];
243
244 // Compute the rhs for velocity
245 for (int i = 0; i < 3; i++) {
246 rhs[3 + i] =
247 -d_lapse_x[i] +
248 (VecVecMul(d_lapse_x, V_up) -
249 lapse_x * VecVecMul(SMatVecMul(curv_x, V_up), V_up)) *
250 V_down[i] +
251 0.5 * lapse_x * VecVecMul(SMatVecMul(d_gamma_x[i], V_up), V_up) +
252 VecVecMul(V_down, d_shift_x[i]);
253 }
254
255 // Compute the rhs for energy
256 rhs[3 + StructType::E] = 0;
257
258 return rhs;
259
260} // PhotonsContainer::compute_rhs

281template <typename StructType>
282void PhotonsContainer<StructType>::evolveRK2(const amrex::MultiFab &lapse,
283 const amrex::MultiFab &shift,
284 const amrex::MultiFab &metric,
285 const amrex::MultiFab &curv,
286 const CCTK_REAL &dt,
287 const int &lev) {
288 CCTK_INFO("Evolving using RK2");
289
290 const auto plo0 = this->Geom(0).ProbLoArray();
291 const auto phi0 = this->Geom(0).ProbHiArray();
292
293 const auto dx = this->Geom(lev).CellSizeArray();
294 const auto plo = this->Geom(lev).ProbLoArray();
295
296 // for each particle:
297 for (Iterator::ParticleIterator<StructType> pti(*this, lev); pti.isValid();
298 ++pti) {
299 const int np = pti.numParticles();
300
301 // Get the information relate to the velocities.
302 auto &attribs = pti.GetAttributes();
303 CCTK_REAL *AMREX_RESTRICT vels_x = attribs[StructType::vx].data();
304 CCTK_REAL *AMREX_RESTRICT vels_y = attribs[StructType::vy].data();
305 CCTK_REAL *AMREX_RESTRICT vels_z = attribs[StructType::vz].data();
306 auto *AMREX_RESTRICT particles = &(pti.GetArrayOfStructs()[0]);
307
308 // Get the arrays of the parameters.
309 auto const lapse_array = lapse.array(pti);
310 auto const shift_array = shift.array(pti);
311 auto const metric_array = metric.array(pti);
312 auto const curv_array = curv.array(pti);
313
314 // foreach particle in the particleContainer
315 amrex::ParallelFor(np, [=] AMREX_GPU_DEVICE(int i) noexcept {
316 // Store the coordinates U to evolve into an array.
317 amrex::GpuArray<CCTK_REAL, StructType::n_attributes + 3> U = {
318 particles[i].pos(0), particles[i].pos(1),
319 particles[i].pos(2), attribs[StructType::vx][i],
320 attribs[StructType::vy][i], attribs[StructType::vz][i],
321 attribs[StructType::E][i]};
322
323 // rhs(u , t) for the runge kutta 2 step
324 auto rhs_1 =
325 this->compute_rhs(U, 0.0, lapse_array, shift_array, metric_array,
326 curv_array, dt, dx, lev, plo);
327
328 U[0] += 0.5 * dt * rhs_1[0];
329 U[1] += 0.5 * dt * rhs_1[1];
330 U[2] += 0.5 * dt * rhs_1[2];
331 U[3] += 0.5 * dt * rhs_1[3];
332 U[4] += 0.5 * dt * rhs_1[4];
333 U[5] += 0.5 * dt * rhs_1[5];
334 U[6] += 0.5 * dt * rhs_1[6];
335
336 // Check if is outside of the grid
337 if (U[0] > (phi0[0] - dx[0]) || U[0] < (plo0[0] + dx[0]) ||
338 U[1] > (phi0[1] - dx[1]) || U[1] < (plo0[1] + dx[1]) ||
339 U[2] > (phi0[2] - dx[2]) || U[2] < (plo0[2] + dx[2])) {
340 particles[i].id() = -1;
341 return;
342 }
343
344 // rhs(u + 0.5 * dt * rhs(u,t), t) for the runge kutta 2 step
345 rhs_1 = this->compute_rhs(U, 0.0 + 0.5 * dt, lapse_array, shift_array,
346 metric_array, curv_array, dt, dx, lev, plo);
347
348 // Runge kutta 2 step
349 particles[i].pos(0) += dt * rhs_1[0];
350 particles[i].pos(1) += dt * rhs_1[1];
351 particles[i].pos(2) += dt * rhs_1[2];
352 vels_x[i] += dt * rhs_1[3];
353 vels_y[i] += dt * rhs_1[4];
354 vels_z[i] += dt * rhs_1[5];
355
356 // Check if is outside of the grid
357 if (particles[i].pos(0) > (phi0[0] - dx[0]) ||
358 particles[i].pos(0) < (plo0[0] + dx[0]) ||
359 particles[i].pos(1) > (phi0[1] - dx[1]) ||
360 particles[i].pos(1) < (plo0[1] + dx[1]) ||
361 particles[i].pos(2) > (phi0[2] - dx[2]) ||
362 particles[i].pos(2) < (plo0[2] + dx[2])) {
363 particles[i].id() = -1;
364 return;
365 }
366 });
367 }
368
369 this->Redistribute();
370} // PhotonsContainer::evolveRK2

401template <typename StructType>
402void PhotonsContainer<StructType>::evolveRK4(const amrex::MultiFab &lapse,
403 const amrex::MultiFab &shift,
404 const amrex::MultiFab &metric,
405 const amrex::MultiFab &curv,
406 const CCTK_REAL &dt,
407 const int &lev) {
408 CCTK_INFO("Evolving particles using Runge-Kutta 4.");
409
410 const auto plo0 = this->Geom(0).ProbLoArray();
411 const auto phi0 = this->Geom(0).ProbHiArray();
412
413 const auto dx = this->Geom(lev).CellSizeArray();
414 const auto plo = this->Geom(lev).ProbLoArray();
415
416 for (Iterator::ParticleIterator<StructType> pti(*this, lev); pti.isValid();
417 ++pti) {
418 const int np = pti.numParticles();
419
420 // Get the information relate to the velocities.
421 auto &attribs = pti.GetAttributes();
422 CCTK_REAL *AMREX_RESTRICT vels_x = attribs[StructType::vx].data();
423 CCTK_REAL *AMREX_RESTRICT vels_y = attribs[StructType::vy].data();
424 CCTK_REAL *AMREX_RESTRICT vels_z = attribs[StructType::vz].data();
425 auto *AMREX_RESTRICT particles = &(pti.GetArrayOfStructs()[0]);
426
427 // Get the array of each parameters.
428 auto const lapse_array = lapse.array(pti);
429 auto const shift_array = shift.array(pti);
430 auto const metric_array = metric.array(pti);
431 auto const curv_array = curv.array(pti);
432
433 amrex::ParallelFor(np, [=] AMREX_GPU_DEVICE(int i) noexcept {
434 amrex::GpuArray<CCTK_REAL, StructType::n_attributes + 3> U = {
435 particles[i].pos(0), particles[i].pos(1),
436 particles[i].pos(2), attribs[StructType::vx][i],
437 attribs[StructType::vy][i], attribs[StructType::vz][i],
438 attribs[StructType::E][i]};
439
440 // f1 = rhs(u , t) for the runge kutta 4 step
441 auto rhs_1 =
442 this->compute_rhs(U, 0.0, lapse_array, shift_array, metric_array,
443 curv_array, dt, dx, lev, plo);
444
445 U[0] += 0.5 * dt * rhs_1[0];
446 U[1] += 0.5 * dt * rhs_1[1];
447 U[2] += 0.5 * dt * rhs_1[2];
448 U[3] += 0.5 * dt * rhs_1[3];
449 U[4] += 0.5 * dt * rhs_1[4];
450 U[5] += 0.5 * dt * rhs_1[5];
451 U[6] += 0.5 * dt * rhs_1[6];
452
453 if (U[0] > (phi0[0] - 0.5 * dx[0]) || U[0] < (plo0[0] + 0.5 * dx[0]) ||
454 U[1] > (phi0[1] - 0.5 * dx[1]) || U[1] < (plo0[1] + 0.5 * dx[1]) ||
455 U[2] > (phi0[2] - 0.5 * dx[2]) || U[2] < (plo0[2] + 0.5 * dx[2])) {
456 particles[i].id() = -1;
457 return;
458 }
459
460 // f2 = rhs(u + 0.5 * dt * f1, t) for the runge kutta 4 step
461 auto rhs_2 =
462 this->compute_rhs(U, 0.0 + 0.5 * dt, lapse_array, shift_array,
463 metric_array, curv_array, dt, dx, lev, plo);
464
465 // Update particles with the f1 and f2 from RK4
466 particles[i].pos(0) += (1. / 6.) * dt * (rhs_1[0] + 2. * rhs_2[0]);
467 particles[i].pos(1) += (1. / 6.) * dt * (rhs_1[1] + 2. * rhs_2[1]);
468 particles[i].pos(2) += (1. / 6.) * dt * (rhs_1[2] + 2. * rhs_2[2]);
469 vels_x[i] += (1. / 6.) * dt * (rhs_1[3] + 2. * rhs_2[3]);
470 vels_y[i] += (1. / 6.) * dt * (rhs_1[4] + 2. * rhs_2[4]);
471 vels_z[i] += (1. / 6.) * dt * (rhs_1[5] + 2. * rhs_2[5]);
472
473 U[0] += 0.5 * dt * rhs_2[0];
474 U[1] += 0.5 * dt * rhs_2[1];
475 U[2] += 0.5 * dt * rhs_2[2];
476 U[3] += 0.5 * dt * rhs_2[3];
477 U[4] += 0.5 * dt * rhs_2[4];
478 U[5] += 0.5 * dt * rhs_2[5];
479 U[6] += 0.5 * dt * rhs_2[6];
480
481 if (particles[i].pos(0) > (phi0[0] - 0.5 * dx[0]) ||
482 particles[i].pos(0) < (plo0[0] + 0.5 * dx[0]) ||
483 particles[i].pos(1) > (phi0[1] - 0.5 * dx[1]) ||
484 particles[i].pos(1) < (plo0[1] + 0.5 * dx[1]) ||
485 particles[i].pos(2) > (phi0[2] - 0.5 * dx[2]) ||
486 particles[i].pos(2) < (plo0[2] + 0.5 * dx[2])) {
487 particles[i].id() = -1;
488 return;
489 }
490
491 if (U[0] > (phi0[0] - 0.5 * dx[0]) || U[0] < (plo0[0] + 0.5 * dx[0]) ||
492 U[1] > (phi0[1] - 0.5 * dx[1]) || U[1] < (plo0[1] + 0.5 * dx[1]) ||
493 U[2] > (phi0[2] - 0.5 * dx[2]) || U[2] < (plo0[2] + 0.5 * dx[2])) {
494 particles[i].id() = -1;
495 return;
496 }
497
498 // f3 = rhs(u + 0.5 * dt * f2, t) for the runge kutta 4 step
499 rhs_1 = this->compute_rhs(U, 0.0, lapse_array, shift_array, metric_array,
500 curv_array, dt, dx, lev, plo);
501
502 U[0] += dt * rhs_1[0];
503 U[1] += dt * rhs_1[1];
504 U[2] += dt * rhs_1[2];
505 U[3] += dt * rhs_1[3];
506 U[4] += dt * rhs_1[4];
507 U[5] += dt * rhs_1[5];
508 U[6] += dt * rhs_1[6];
509
510 if (U[0] > (phi0[0] - 0.5 * dx[0]) || U[0] < (plo0[0] + 0.5 * dx[0]) ||
511 U[1] > (phi0[1] - 0.5 * dx[1]) || U[1] < (plo0[1] + 0.5 * dx[1]) ||
512 U[2] > (phi0[2] - 0.5 * dx[2]) || U[2] < (plo0[2] + 0.5 * dx[2])) {
513 particles[i].id() = -1;
514 return;
515 }
516
517 // f4 = rhs(u + dt * f3, t) for the runge kutta 4 step
518 rhs_2 = this->compute_rhs(U, 0.0, lapse_array, shift_array, metric_array,
519 curv_array, dt, dx, lev, plo);
520
521 // Update particles with the f3 and f4 from RK4
522 particles[i].pos(0) += (1. / 6.) * dt * (2. * rhs_1[0] + rhs_2[0]);
523 particles[i].pos(1) += (1. / 6.) * dt * (2. * rhs_1[1] + rhs_2[1]);
524 particles[i].pos(2) += (1. / 6.) * dt * (2. * rhs_1[2] + rhs_2[2]);
525 vels_x[i] += (1. / 6.) * dt * (2. * rhs_1[3] + rhs_2[3]);
526 vels_y[i] += (1. / 6.) * dt * (2. * rhs_1[4] + rhs_2[4]);
527 vels_z[i] += (1. / 6.) * dt * (2. * rhs_1[5] + rhs_2[5]);
528
529 // Check if is outside of the grid
530 if (particles[i].pos(0) > (phi0[0] - 0.5 * dx[0]) ||
531 particles[i].pos(0) < (plo0[0] + 0.5 * dx[0]) ||
532 particles[i].pos(1) > (phi0[1] - 0.5 * dx[1]) ||
533 particles[i].pos(1) < (plo0[1] + 0.5 * dx[1]) ||
534 particles[i].pos(2) > (phi0[2] - 0.5 * dx[2]) ||
535 particles[i].pos(2) < (plo0[2] + 0.5 * dx[2])) {
536 particles[i].id() = -1;
537 return;
538 }
539 });
540 }

542 this->Redistribute();

553template <typename StructType>
554void PhotonsContainer<StructType>::check_velocity(CCTK_ARGUMENTS,
555 const amrex::MultiFab &metric,
556 const int &lev) {
557 // Get the current iteration number.
558 const int iteration = cctkGH->cctk_iteration;
559 CCTK_VINFO("Checking the velocity error at iteration %d.", iteration);
560
561 // Initializing the files.
562 std::ostringstream file_name;
563 file_name << "Velocity_" << iteration << ".dat";
564 std::ofstream vel_file;
565 vel_file.open(file_name.str());
566
567 const auto plo0 = this->Geom(0).ProbLoArray();
568 const auto phi0 = this->Geom(0).ProbHiArray();
569
570 const auto dx = this->Geom(lev).CellSizeArray();
571 const auto p_lo = this->Geom(lev).ProbLoArray();
572
573 for (Iterator::ParticleIterator<StructType> pti(*this, lev); pti.isValid();
574 ++pti) {
575 const int np = pti.numParticles();
576
577 // Get the velocity information
578 auto &attribs = pti.GetAttributes();
579 CCTK_REAL *AMREX_RESTRICT vels_x = attribs[StructType::vx].data();
580 CCTK_REAL *AMREX_RESTRICT vels_y = attribs[StructType::vy].data();
581 CCTK_REAL *AMREX_RESTRICT vels_z = attribs[StructType::vz].data();
582 auto *AMREX_RESTRICT particles = &(pti.GetArrayOfStructs()[0]);
583
584 auto const metric_array = metric.array(pti);
585
586 // foreach particle
587 amrex::ParallelFor(np, [=, &vel_file] AMREX_GPU_DEVICE(int i) noexcept {
588 const auto p = particles[i];
589
590 // Get the cell's index at the particle's position.
591 const int i0 = amrex::Math::floor((p.pos(0) - p_lo[0]) / dx[0]);
592 const int j0 = amrex::Math::floor((p.pos(1) - p_lo[1]) / dx[1]);
593 const int k0 = amrex::Math::floor((p.pos(2) - p_lo[2]) / dx[2]);
594
595 // Interpolate metric
596 const amrex::GpuArray<CCTK_REAL, 6> gamma_x = {
597 barycentric_cubic_3d<3>(metric_array, i0, j0, k0, p.pos(0), p.pos(1),
598 p.pos(2), dx, p_lo,
599 0), // g_11
600 barycentric_cubic_3d<3>(metric_array, i0, j0, k0, p.pos(0), p.pos(1),
601 p.pos(2), dx, p_lo,
602 1), // g_12 & g_21
603 barycentric_cubic_3d<3>(metric_array, i0, j0, k0, p.pos(0), p.pos(1),
604 p.pos(2), dx, p_lo,
605 2), // g_13 & g_31
606 barycentric_cubic_3d<3>(metric_array, i0, j0, k0, p.pos(0), p.pos(1),
607 p.pos(2), dx, p_lo,
608 3), // g_22
609 barycentric_cubic_3d<3>(metric_array, i0, j0, k0, p.pos(0), p.pos(1),
610 p.pos(2), dx, p_lo,
611 4), // g_23, g_32
612 barycentric_cubic_3d<3>(metric_array, i0, j0, k0, p.pos(0), p.pos(1),
613 p.pos(2), dx, p_lo, 5)}; // g_33
614
615 // Get the inverse metric.
616 const CCTK_REAL inv_det_gamma =
617 1.0 / (gamma_x[0] * gamma_x[3] * gamma_x[5] +
618 2. * gamma_x[1] * gamma_x[2] * gamma_x[4] -
619 gamma_x[2] * gamma_x[2] * gamma_x[3] -
620 gamma_x[4] * gamma_x[4] * gamma_x[0] -
621 gamma_x[1] * gamma_x[1] * gamma_x[5]);
622
623 const amrex::GpuArray<CCTK_REAL, 6> gamma_inv_x = {
624 (gamma_x[3] * gamma_x[5] - gamma_x[4] * gamma_x[4]) * inv_det_gamma,
625 (gamma_x[4] * gamma_x[2] - gamma_x[1] * gamma_x[5]) * inv_det_gamma,
626 (gamma_x[1] * gamma_x[4] - gamma_x[2] * gamma_x[3]) * inv_det_gamma,
627 (gamma_x[0] * gamma_x[5] - gamma_x[2] * gamma_x[2]) * inv_det_gamma,
628 (gamma_x[2] * gamma_x[1] - gamma_x[0] * gamma_x[4]) * inv_det_gamma,
629 (gamma_x[0] * gamma_x[3] - gamma_x[1] * gamma_x[1]) * inv_det_gamma};
630
631 // Get the V^2 of each photon
632 const CCTK_REAL v_squared = vels_x[i] * vels_x[i] * gamma_inv_x[0] +
633 vels_y[i] * vels_y[i] * gamma_inv_x[3] +
634 vels_z[i] * vels_z[i] * gamma_inv_x[5] +
635 2.0 * vels_x[i] * vels_y[i] * gamma_inv_x[1] +
636 2.0 * vels_x[i] * vels_z[i] * gamma_inv_x[2] +
637 2.0 * vels_y[i] * vels_z[i] * gamma_inv_x[4];
638
639 // Write it into a file.
640 vel_file << p.id() << "\t" << std::abs(v_squared - 1.0) << "\n";
641 });
642 }
643
644 vel_file.close();
645
646} // PhotonsContainer::check_velocity
