165 void check_banned_zones(const int &level, const CCTK_INT4 &zones,
166 const CCTK_REAL (&x)[10], const CCTK_REAL (&y)[10],
167 const CCTK_REAL (&z)[10],
168 const CCTK_REAL (&radius)[10]) {
169
170 if (!zones) {
171 return;
172 }
173
174 for (Iterator::ParticleIterator<StructType> pti(*this, level);
175 pti.isValid(); ++pti) {
176 const int np = pti.numParticles();
177 auto *AMREX_RESTRICT particles = &(pti.GetArrayOfStructs()[0]);
178
179 auto self = this;
180 amrex::ParallelFor(np, [=] AMREX_GPU_DEVICE(int i) noexcept {
181 bool out = false;
182 for (int check = 0; check < zones; check++) {
183 const CCTK_REAL dx = particles[i].pos(0) - x[check];
184 const CCTK_REAL dy = particles[i].pos(1) - y[check];
185 const CCTK_REAL dz = particles[i].pos(2) - z[check];
186 out |= (dx * dx + dy * dy + dz * dz <= radius[check] * radius[check]);
187 }
188
189 if (out) {
190 particles[i].id() = -1;
191 return;
192 }
193 });
194 }
195 }
