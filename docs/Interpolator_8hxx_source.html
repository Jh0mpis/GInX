51template <int N>
52AMREX_GPU_HOST_DEVICE AMREX_FORCE_INLINE CCTK_ATTRIBUTE_ALWAYS_INLINE void
53barycentric_cubic_1d(CCTK_REAL &value, const int (&points)[N],
54 const CCTK_REAL *weights, const CCTK_REAL (&values)[N],
55 const CCTK_REAL &x, const CCTK_REAL &plo,
56 const CCTK_REAL &dx) {
57 CCTK_REAL num{0.0};
58 CCTK_REAL den{0.0};
59
60 for (int i = 0; i < N; i++) {
61 // Check if the point belongs to the consulted points.
62 CCTK_REAL diff = x - (plo + points[i] * dx);
63 const CCTK_REAL tolerance = 1e-12;
64 if (diff < tolerance && diff > -tolerance) {
65 value = values[i];
66 return;
67 }
68 // Compute the weights and values
69 CCTK_REAL term = weights[i] / diff;
70 num += term * values[i];
71 den += term;
72 }
73
74 // Return the interpolation
75 value = num / den;
76}

262template <int N>
263AMREX_GPU_DEVICE
264 AMREX_GPU_HOST AMREX_FORCE_INLINE CCTK_ATTRIBUTE_ALWAYS_INLINE void
265 der_barycentric_cubic_1d(CCTK_REAL &f_x, CCTK_REAL &d_f_x,
266 const int (&points)[N], const CCTK_REAL *weights,
267 const CCTK_REAL (&values)[N], const CCTK_REAL &x,
268 const CCTK_REAL &plo, const CCTK_REAL &dx) {
269 CCTK_REAL num{0.0};
270 CCTK_REAL den{0.0};
271 CCTK_REAL den_sqr{0.0};
272 CCTK_REAL der_num{0.0};
273 d_f_x = 0.0;
274
275 // Compute the interpolation
276 for (int i = 0; i < N; i++) {
277 CCTK_REAL diff = x - (plo + points[i] * dx);
278 const CCTK_REAL tolerance = 1e-12;
279 if (diff < tolerance && diff > -tolerance) {
280 // Check if the point makes part of the points used on the
281 // interpolation.
282 for (int j = 0; j < N; j++) {
283 if (i == j) {
284 continue;
285 }
286 const auto x_j = points[j] * dx;
287 const auto x_i = points[i] * dx;
288 d_f_x += weights[j] * (values[i] - values[j]) / (x_j - x_i);
289 }
290 d_f_x /= weights[i];
291 f_x = values[i];
292 return;
293 }
294
295 // Compute the weights for the interpolation and the derivative.
296 CCTK_REAL term = weights[i] / diff;
297 num += term * values[i];
298 den += term;
299 den_sqr += term / diff;
300 }
301
302 // Compute the weights for the derivative.
303 for (int i = 0; i < N; i++) {
304 CCTK_REAL term = weights[i] / (x - (plo + points[i] * dx));
305 der_num += (-term * den / (x - (plo + points[i] * dx)) + term * den_sqr) *
306 values[i];
307 }
308
309 // Fill the values
310 f_x = num / den;
311 d_f_x = der_num / (den * den);
312} // der_barycentric_cubic_1d
