68template <int N>
69AMREX_GPU_DEVICE AMREX_GPU_HOST CCTK_ATTRIBUTE_ALWAYS_INLINE inline CCTK_REAL
70barycentric_cubic_1d(const int (&points)[N], const CCTK_REAL *weights,
71 const CCTK_REAL (&values)[N], const CCTK_REAL &x,
72 const CCTK_REAL &plo, const CCTK_REAL &dx) {
73 CCTK_REAL num{0.0};
74 CCTK_REAL den{0.0};
75
76 for (int i = 0; i < N; i++) {
77 // Check if the point belongs to the consulted points.
78 CCTK_REAL diff = x - (plo + points[i] * dx);
79 const CCTK_REAL tolerance = 1e-12;
80 if (diff < tolerance && diff > -tolerance) {
81 return values[i];
82 }
83 // Compute the weights and values
84 CCTK_REAL term = weights[i] / diff;
85 num += term * values[i];
86 den += term;
87 }
88
89 // Return the interpolation
90 return num / den;
91}

128template <int INTERPOLATION_ORDER>
129AMREX_GPU_HOST_DEVICE CCTK_ATTRIBUTE_ALWAYS_INLINE inline CCTK_REAL
130barycentric_cubic_3d(amrex::Array4<CCTK_REAL const> const &f, int const &i0,
131 int const &j0, int const &k0, CCTK_REAL const &x,
132 CCTK_REAL const &y, CCTK_REAL const &z,
133 const amrex::GpuArray<double, 3> &dx,
134 const amrex::GpuArray<double, 3> &plo, const int &comp) {
135 const int order =
136 (((INTERPOLATION_ORDER - 1) * INTERPOLATION_ORDER) >> 1) - 1;
137 const CCTK_INT *nodes = &all_nodes[order];
138 const CCTK_REAL *w = &all_weights[order];
139
140 // Setting the pre-computed weights.
141 if constexpr (INTERPOLATION_ORDER > 5 || INTERPOLATION_ORDER < 2) {
142 CCTK_INFO("Barycentric Lagrange interpolation of desired order is not yet "
143 "implemented. Available orders: 1, 2, 3.");
144 throw std::invalid_argument(
145 "Wrong order of barycentric Lagrange interpolation");
146 }
147
148 // Do the interpolation on x
149 CCTK_REAL G[INTERPOLATION_ORDER][INTERPOLATION_ORDER];
150 for (int j = 0; j < INTERPOLATION_ORDER; j++) {
151 for (int k = 0; k < INTERPOLATION_ORDER; k++) {
152 CCTK_REAL values[INTERPOLATION_ORDER];
153 CCTK_INT points[INTERPOLATION_ORDER];
154 for (int i = 0; i < INTERPOLATION_ORDER; i++) {
155 values[i] = f(i0 + nodes[i], j0 + nodes[j], k0 + nodes[k], comp);
156 points[i] = i0 + nodes[i];
157 }
158 G[j][k] = barycentric_cubic_1d<INTERPOLATION_ORDER>(points, w, values, x,
159 plo[0], dx[0]);
160 }
161 }
162
163 // Do the interpolation on y
164 CCTK_REAL H[INTERPOLATION_ORDER];
165 for (int k = 0; k < INTERPOLATION_ORDER; k++) {
166 CCTK_REAL values[INTERPOLATION_ORDER];
167 CCTK_INT points[INTERPOLATION_ORDER];
168 for (int j = 0; j < INTERPOLATION_ORDER; j++) {
169 values[j] = G[j][k];
170 points[j] = j0 + nodes[j];
171 }
172 H[k] = barycentric_cubic_1d<INTERPOLATION_ORDER>(points, w, values, y,
173 plo[1], dx[1]);
174 }
175
176 // Do the interpolation on z
177 CCTK_INT points[INTERPOLATION_ORDER];
178 for (int k = 0; k < INTERPOLATION_ORDER; k++) {
179 points[k] = k0 + nodes[k];
180 }
181 return barycentric_cubic_1d<INTERPOLATION_ORDER>(points, w, H, z, plo[2],
182 dx[2]);
183} // barycentric_cubic_3d with component

268template <int N>
269AMREX_GPU_DEVICE AMREX_GPU_HOST CCTK_ATTRIBUTE_ALWAYS_INLINE inline void
270der_barycentric_cubic_1d(CCTK_REAL &f_x, CCTK_REAL &d_f_x,
271 const int (&points)[N], const CCTK_REAL *weights,
272 const CCTK_REAL (&values)[N], const CCTK_REAL &x,
273 const CCTK_REAL &plo, const CCTK_REAL &dx) {
274 CCTK_REAL num{0.0};
275 CCTK_REAL den{0.0};
276 CCTK_REAL den_sqr{0.0};
277 CCTK_REAL der_num{0.0};
278 d_f_x = 0.0;
279
280 // Compute the interpolation
281 for (int i = 0; i < N; i++) {
282 CCTK_REAL diff = x - (plo + points[i] * dx);
283 const CCTK_REAL tolerance = 1e-12;
284 if (diff < tolerance && diff > -tolerance) {
285 // Check if the point makes part of the points used on the
286 // interpolation.
287 for (int j = 0; j < N; j++) {
288 if (i == j) {
289 continue;
290 }
291 const auto x_j = points[j] * dx;
292 const auto x_i = points[i] * dx;
293 d_f_x += weights[j] * (values[i] - values[j]) / (x_j - x_i);
294 // CCTK_VWarn(1, __LINE__, __FILE__, CCTK_THORNSTRING, "d_f_x %f %f %f",
295 // x, weights[j] * (values[i] - values[j]) , (x_j - x_i));
296 }
297 d_f_x /= weights[i];
298 f_x = values[i];
299 return;
300 }
301
302 // Compute the weights for the interpolation and the derivative.
303 CCTK_REAL term = weights[i] / diff;
304 num += term * values[i];
305 den += term;
306 den_sqr += term / diff;
307 }
308
309 // Compute the weights for the derivative.
310 for (int i = 0; i < N; i++) {
311 CCTK_REAL term = weights[i] / (x - (plo + points[i] * dx));
312 der_num += (-term * den / (x - (plo + points[i] * dx)) + term * den_sqr) *
313 values[i];
314 }
315
316 // Fill the values
317 f_x = num / den;
318 d_f_x = der_num / (den * den);
319} // der_barycentric_cubic_1d
