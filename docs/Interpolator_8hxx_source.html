56template <int N>
57AMREX_GPU_DEVICE AMREX_GPU_HOST CCTK_ATTRIBUTE_ALWAYS_INLINE inline CCTK_REAL
58barycentric_cubic_1d(const int (&points)[N],
59 const CCTK_REAL *weights,
60 const CCTK_REAL (&values)[N], const CCTK_REAL &x,
61 const CCTK_REAL &plo, const CCTK_REAL &dx) {
62 CCTK_REAL num{0.0};
63 CCTK_REAL den{0.0};
64
65 for (int i = 0; i < N; i++) {
66 // Check if the point belongs to the consulted points.
67 CCTK_REAL diff = x - (plo + points[i] * dx);
68 const CCTK_REAL tolerance = 1e-12;
69 if (diff < tolerance && diff > -tolerance) {
70 return values[i];
71 }
72 // Compute the weights and values
73 CCTK_REAL term = weights[i] / diff;
74 num += term * values[i];
75 den += term;
76 }
77
78 // Return the interpolation
79 return num / den;
80}

117template <int INTERPOLATION_ORDER>
118AMREX_GPU_HOST_DEVICE CCTK_ATTRIBUTE_ALWAYS_INLINE inline CCTK_REAL
119barycentric_cubic_3d(amrex::Array4<CCTK_REAL const> const &f, int const &i0,
120 int const &j0, int const &k0, CCTK_REAL const &x,
121 CCTK_REAL const &y, CCTK_REAL const &z,
122 const amrex::GpuArray<double, 3> &dx,
123 const amrex::GpuArray<double, 3> &plo, const int &comp) {
124 const int order = (((INTERPOLATION_ORDER - 1) * INTERPOLATION_ORDER) >> 1) - 1;
125 const CCTK_INT *nodes = &all_nodes[order];
126 const CCTK_REAL *w = &all_weights[order];
127
128 // Setting the pre-computed weights.
129 if constexpr (INTERPOLATION_ORDER > 5 || INTERPOLATION_ORDER < 2) {
130 CCTK_INFO("Barycentric Lagrange interpolation of desired order is not yet "
131 "implemented. Available orders: 1, 2, 3.");
132 throw std::invalid_argument(
133 "Wrong order of barycentric Lagrange interpolation");
134 }
135
136 // Do the interpolation on x
137 CCTK_REAL G[INTERPOLATION_ORDER][INTERPOLATION_ORDER];
138 for (int j = 0; j < INTERPOLATION_ORDER; j++) {
139 for (int k = 0; k < INTERPOLATION_ORDER; k++) {
140 CCTK_REAL values[INTERPOLATION_ORDER];
141 CCTK_INT points[INTERPOLATION_ORDER];
142 for (int i = 0; i < INTERPOLATION_ORDER; i++) {
143 values[i] = f(i0 + nodes[i], j0 + nodes[j], k0 + nodes[k], comp);
144 points[i] = i0 + nodes[i];
145 }
146 G[j][k] = barycentric_cubic_1d<INTERPOLATION_ORDER>(points, w, values, x,
147 plo[0], dx[0]);
148 }
149 }
150
151 // Do the interpolation on y
152 CCTK_REAL H[INTERPOLATION_ORDER];
153 for (int k = 0; k < INTERPOLATION_ORDER; k++) {
154 CCTK_REAL values[INTERPOLATION_ORDER];
155 CCTK_INT points[INTERPOLATION_ORDER];
156 for (int j = 0; j < INTERPOLATION_ORDER; j++) {
157 values[j] = G[j][k];
158 points[j] = j0 + nodes[j];
159 }
160 H[k] = barycentric_cubic_1d<INTERPOLATION_ORDER>(points, w, values, y,
161 plo[1], dx[1]);
162 }
163
164 // Do the interpolation on z
165 CCTK_INT points[INTERPOLATION_ORDER];
166 for (int k = 0; k < INTERPOLATION_ORDER; k++) {
167 points[k] = k0 + nodes[k];
168 }
169 return barycentric_cubic_1d<INTERPOLATION_ORDER>(points, w, H, z, plo[2],
170 dx[2]);
171} // barycentric_cubic_3d with component

256template <int N>
257AMREX_GPU_DEVICE AMREX_GPU_HOST CCTK_ATTRIBUTE_ALWAYS_INLINE inline void
258der_barycentric_cubic_1d(CCTK_REAL &f_x, CCTK_REAL &d_f_x,
259 const int (&points)[N],
260 const CCTK_REAL *weights,
261 const CCTK_REAL (&values)[N], const CCTK_REAL &x,
262 const CCTK_REAL &plo, const CCTK_REAL &dx) {
263 CCTK_REAL num{0.0};
264 CCTK_REAL den{0.0};
265 CCTK_REAL den_sqr{0.0};
266 CCTK_REAL der_num{0.0};
267
268 // Compute the interpolation
269 for (int i = 0; i < N; i++) {
270 CCTK_REAL diff = x - (plo + points[i] * dx);
271 const CCTK_REAL tolerance = 1e-12;
272 if (diff < tolerance && diff > -tolerance) {
273 // Check if the point makes part of the points used on the
274 // interpolation.
275 for (int j = 0; j <= N; j++) {
276 if (i == j) {
277 continue;
278 }
279 const auto x_j = points[j] * dx;
280 const auto x_i = points[i] * dx;
281 d_f_x += weights[j] * (values[i] - values[j]) / (x_j - x_i);
282 }
283 d_f_x /= weights[i];
284 f_x = values[i];
285 return;
286 }
287
288 // Compute the weights for the interpolation and the derivative.
289 CCTK_REAL term = weights[i] / diff;
290 num += term * values[i];
291 den += term;
292 den_sqr += term / diff;
293 }
294
295 // Compute the weights for the derivative.
296 for (int i = 0; i < N; i++) {
297 CCTK_REAL term = weights[i] / (x - (plo + points[i] * dx));
298 der_num += (-term * den / (x - (plo + points[i] * dx)) + term * den_sqr) *
299 values[i];
300 }
301
302 // Fill the values
303 f_x = num / den;
304 d_f_x = der_num / (den * den);
305} // der_barycentric_cubic_1d
