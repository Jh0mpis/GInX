52template <int N>
53AMREX_GPU_DEVICE AMREX_GPU_HOST CCTK_ATTRIBUTE_ALWAYS_INLINE inline CCTK_REAL
54barycentric_cubic_1d(const int (&points)[N + 1],
55 const CCTK_REAL (&weights)[N + 1],
56 const CCTK_REAL (&values)[N + 1], const CCTK_REAL &x,
57 const CCTK_REAL &plo, const CCTK_REAL &dx) {
58 CCTK_REAL num{0.0};
59 CCTK_REAL den{0.0};
60
61 for (int i = 0; i <= N; i++) {
62 // Check if the point belongs to the consulted points.
63 if (x == plo + points[i] * dx) {
64 return values[i];
65 }
66 // Compute the weights and values
67 CCTK_REAL term = weights[i] / (x - (plo + points[i] * dx));
68 num += term * values[i];
69 den += term;
70 }
71
72 // Return the interpolation
73 return num / den;
74}

110template <int INTERPOLATION_ORDER>
111AMREX_GPU_HOST_DEVICE CCTK_ATTRIBUTE_ALWAYS_INLINE inline CCTK_REAL
112barycentric_cubic_3d(amrex::Array4<CCTK_REAL const> const &f, int const &i0,
113 int const &j0, int const &k0, CCTK_REAL const &x,
114 CCTK_REAL const &y, CCTK_REAL const &z,

120 CCTK_REAL w[INTERPOLATION_ORDER + 1];

144 nodes[0] = -2;
145 nodes[1] = -1;
146 nodes[2] = 0;
147 nodes[3] = 1;
148 nodes[4] = 2;

151 w[2] = 0.25;
152 w[3] = (-1.0 / 6.0);
153 w[4] = (1.0 / 24.0);

159 }

166 CCTK_INT points[INTERPOLATION_ORDER + 1];
167 for (int i = 0; i <= INTERPOLATION_ORDER; i++) {

181 for (int j = 0; j <= INTERPOLATION_ORDER; j++) {

191 for (int k = 0; k <= INTERPOLATION_ORDER; k++) {
192 points[k] = k0 + nodes[k];

280template <int N>
281AMREX_GPU_DEVICE AMREX_GPU_HOST CCTK_ATTRIBUTE_ALWAYS_INLINE inline void
282der_barycentric_cubic_1d(CCTK_REAL &f_x, CCTK_REAL &d_f_x,
283 const int (&points)[N + 1],
284 const CCTK_REAL (&weights)[N + 1],
285 const CCTK_REAL (&values)[N + 1], const CCTK_REAL &x,
286 const CCTK_REAL &plo, const CCTK_REAL &dx) {
287 CCTK_REAL num{0.0};
288 CCTK_REAL den{0.0};
289 CCTK_REAL den_sqr{0.0};
290 CCTK_REAL der_num{0.0};
291
292 // Compute the interpolation
293 for (int i = 0; i <= N; i++) {
294 if (x == plo + points[i] * dx) {
295 // Check if the point makes part of the points used on the interpolation.
296 for (int j = 0; j <= N; j++) {
297 if (i == j) {
298 continue;
299 d_f_x +=
300 weights[j] * (values[i] - values[j]) / (x - plo + points[j] * dx);
301 }
302 }
303 d_f_x /= weights[i];
304 f_x = values[i];
305 d_f_x = 0.0;
306 return;
307 }
308
309 // Compute the weights for the interpolation and the derivative.
310 CCTK_REAL term = weights[i] / (x - (plo + points[i] * dx));
311 num += term * values[i];
312 den += term;
313 den_sqr += term / (x - (plo + points[i] * dx));
314 }
315
316 // Compute the weights for the derivative.
317 for (int i = 0; i <= N; i++) {
318 CCTK_REAL term = weights[i] / (x - (plo + points[i] * dx));
319 der_num += (-term * den / (x - (plo + points[i] * dx)) + term * den_sqr) *
320 values[i];
321 }
322
323 // Fill the values
324 f_x = num / den;
325 d_f_x = der_num / (den * den);
326} // der_barycentric_cubic_1d
