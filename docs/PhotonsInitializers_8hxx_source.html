43template <typename StructType, typename ParticleContainerClass>
44void random_spherical_initializer(ParticleContainerClass &pc,
45 const CCTK_REAL *real_params,
46 const CCTK_INT *int_params) {
47
48 CCTK_INFO("Initializing particles using the "
49 "random_spherical_photons_per_container_initializer");
50
51 const CCTK_INT level = 0;
52 const CCTK_REAL bh_mass = real_params[0];
53
54 // Get the with of the discretization on each direction.
55 const auto dx = pc.Geom(level).CellSizeArray();
56
57 // Get the lower and higher value over the ParticleContainer Geometry
58 const auto p_lo = pc.Geom(level).ProbLoArray();
59 const auto p_hi = pc.Geom(level).ProbHiArray();
60
61 const int n_procs = amrex::ParallelDescriptor::NProcs();
62 const int proc_id = amrex::ParallelDescriptor::MyProc();
63
64 const int total_particles = int_params[0];
65 const int local_particles_size =
66 total_particles / n_procs + (proc_id < total_particles % n_procs);
67
68 int total_tiles{0};
69 for (amrex::MFIter mfi = pc.MakeMFIter(level); mfi.isValid(); ++mfi) {
70 total_tiles++;
71 }
72
73 int current_tile = 0;
74
75 // Iterating over all the tiles of the particle data structure
76 for (amrex::MFIter mfi = pc.MakeMFIter(level); mfi.isValid(); ++mfi) {
77
78 const unsigned int particles_per_tile =
79 local_particles_size / total_tiles +
80 (current_tile < local_particles_size % total_tiles);
81
82 // get each tile box
83 const amrex::Box &tile_box = mfi.tilebox();
84
85 // Get the tile bounds
86 const auto lo = amrex::lbound(tile_box);
87 const auto hi = amrex::ubound(tile_box);
88 // Get a reference to the particles
89 auto &particles = pc.GetParticles(level);
90 auto &particle_tile = pc.DefineAndReturnParticleTile(level, mfi);
91
92 // Determines the current size and the required new size
93 auto old_size = particle_tile.GetArrayOfStructs().size();
94
95 // Resize the container once, we do not need to do it one by one
96 auto new_size = old_size + particles_per_tile;
97 particle_tile.resize(new_size);
98
99 // Gets raw pointers to the two different ways particle data is stored for
100 // performance reasons: Array of Struct (AoS) and Struct of Arrays (SoA)
101 auto *p_struct = particle_tile.GetArrayOfStructs()().data();
102 auto arrdata = particle_tile.GetStructOfArrays().realarray();
103
104 std::vector<long> particle_ids(particles_per_tile);
105 for (int i = 0; i < particles_per_tile; ++i) {
106 particle_ids[i] = ParticleContainerClass::ParticleType::NextID();
107 }
108
109 // Copy IDs to GPU (AMReX handles this automatically if needed)
110 amrex::Gpu::DeviceVector<long> d_particle_ids(particle_ids.size());
111 amrex::Gpu::copyAsync(amrex::Gpu::hostToDevice, particle_ids.begin(),
112 particle_ids.end(), d_particle_ids.begin());
113 const long *id_ptr = d_particle_ids.data();
114
115 amrex::ParallelForRNG(
116 particles_per_tile,
117 [=] AMREX_GPU_DEVICE(int i,
118 amrex::RandomEngine const &engine) noexcept {
119 // Start a for loop with Random Number evolution
120 int pidx = old_size + i;
121
122 // Generate a random position
123 const amrex::Real ratio[AMREX_SPACEDIM] = {
124 (std::abs(p_hi[0] - p_lo[0]) - (bh_mass * 0.5 + 0.2) * 2.) * 0.5 *
125 Random(engine) +
126 bh_mass * 0.5 + 0.2,
127 Random(engine) * M_PI, Random(engine) * 2. * M_PI};
128
129 auto &p = p_struct[pidx];
130
131 p.id() = id_ptr[i];
132 p.cpu() = proc_id;
133
134 p.pos(0) = ratio[0] * std::sin(ratio[1]) * std::cos(ratio[2]);
135 p.pos(1) = ratio[0] * std::sin(ratio[1]) * std::sin(ratio[2]);
136 p.pos(2) = ratio[0] * std::cos(ratio[1]);
137
138 // Create the particle and add it to the container
139 arrdata[StructType::vx][pidx] = 2. * Random(engine) - 1.0;
140 arrdata[StructType::vy][pidx] = 2. * Random(engine) - 1.0;
141 arrdata[StructType::vz][pidx] = 2. * Random(engine) - 1.0;
142 arrdata[StructType::ln_E][pidx] = 0;
143 });
144 CCTK_VWarn(1, __LINE__, __FILE__, CCTK_THORNSTRING,
145 "Number of particles created at tile %d: %d ", current_tile,
146 particles_per_tile);
147 current_tile++;
148 }
149
150 pc.Redistribute();
151 pc.SortParticlesByCell();
152
153 CCTK_VINFO("%d particles created", pc.TotalNumberOfParticles());
154
155} // random_spherical_initializer
