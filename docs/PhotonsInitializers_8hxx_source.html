42template <typename StructType, typename ParticleContainerClass>
43void random_photons_initializer(ParticleContainerClass &pc,
44 const int &number_of_particles_per_cell,
45 const amrex::MultiFab &metric, const int &lev) {
46
47 CCTK_INFO("Initializing particles using the random_photons_initializer");
48
49 // Get the with of the discretization on each direction.
50 const auto dx = pc.Geom(lev).CellSizeArray();
51
52 // Get the lower and higher value over the ParticleContainer Geometry
53 const auto p_lo = pc.Geom(lev).ProbLoArray();
54 const auto p_hi = pc.Geom(lev).ProbHiArray();
55
56 amrex::MFIter mfi = pc.MakeMFIter(lev);
57
58 // Iterating over all the tiles of the particle data structure
59 for (; mfi.isValid(); ++mfi) {
60
61 // get each tile box
62 const amrex::Box &tile_box = mfi.tilebox();
63
64 // Get the tile bounds
65 const auto lo = amrex::lbound(tile_box);
66 const auto hi = amrex::ubound(tile_box);
67
68 // Get a reference to the particles
69 auto &particles = pc.GetParticles(lev);
70 auto &particle_tile =
71 particles[std::make_pair(mfi.index(), mfi.LocalTileIndex())];
72
73 // Determines the current size and the required new size
74 auto old_size = particle_tile.GetArrayOfStructs().size();
75 auto new_size = old_size + number_of_particles_per_cell *
76 (hi.x - lo.x + 1) * (hi.z - lo.z + 1) *
77 (hi.y - lo.y + 1);
78
79 // Resize the container once, we do not need to do it one by one
80 particle_tile.resize(new_size);
81
82 // Gets raw pointers to the two different ways particle data is stored for
83 // performance reasons: Array of Struct (AoS) and Struct of Arrays (SoA)
84 typename ParticleContainerClass::ParticleType *p_struct =
85 particle_tile.GetArrayOfStructs()().data();
86 auto arrdata = particle_tile.GetStructOfArrays().realarray();
87
88 // get the current process id
89 int proc_id = amrex::ParallelDescriptor::MyProc();
90 auto const metric_array = metric.array(mfi);
91
92 // Start a for loop with Random Number evolution
93 amrex::ParallelForRNG(tile_box, [=] AMREX_GPU_DEVICE(
94 int i, int j, int k,
95 amrex::RandomEngine const
96 &engine) noexcept {
97 // Calculate cell_id
98 int ix = i - lo.x;
99 int iy{0}, iz{0};
100
101 int nx = hi.x - lo.x + 1;
102 int ny{0}, nz{0};
103
104 unsigned int uix = amrex::min(nx - 1, amrex::max(0, ix));
105 unsigned int uiy{0}, uiz{0};
106
107 iy = j - lo.y;
108 ny = hi.y - lo.y + 1;
109 uiy = amrex::min(ny - 1, amrex::max(0, iy));
110 iz = k - lo.z;
111 nz = hi.z - lo.z + 1;
112 uiz = amrex::min(nz - 1, amrex::max(0, iz));
113
114 unsigned int cell_id = (uix * ny + uiy) * nz + uiz;
115
116 // Retrievers the starting write index (pidx) for the current cell
117 // (i, j, k) from the offsets array that was calculated by the
118 // exclusive_scan
119 int pidx = old_size + number_of_particles_per_cell * cell_id;
120
121 for (int i_part = 0; i_part < number_of_particles_per_cell; i_part++) {
122 amrex::Real ratio[AMREX_SPACEDIM];
123
124 // Create particles outside of an sphere of radius 0.5
125 // Generate a random position
126 ratio[0] =
127 (std::abs(p_hi[0] - p_lo[0]) * 0.5 - 0.7) * amrex::Random(engine) +
128 0.7;
129 ratio[1] = amrex::Random(engine) * M_PI;
130 ratio[2] = amrex::Random(engine) * 2. * M_PI;
131
132 typename ParticleContainerClass::ParticleType &p = p_struct[pidx];
133 p.id() = pidx + 1;
134 p.cpu() = proc_id;
135
136 p.pos(0) = ratio[0] * std::sin(ratio[1]) * std::cos(ratio[2]);
137 p.pos(1) = ratio[0] * std::sin(ratio[1]) * std::sin(ratio[2]);
138 p.pos(2) = ratio[0] * std::cos(ratio[1]);
139
140 const int i0 = amrex::Math::floor((p.pos(0) - p_lo[0]) / dx[0]);
141 const int j0 = amrex::Math::floor((p.pos(1) - p_lo[1]) / dx[1]);
142 const int k0 = amrex::Math::floor((p.pos(2) - p_lo[2]) / dx[2]);
143
144 // Interpolate metric
145 const amrex::GpuArray<CCTK_REAL, 6> gamma_x = {
146 barycentric_cubic_3d<4>(metric_array, i0, j0, k0, p.pos(0),
147 p.pos(1), p.pos(2), dx, p_lo,
148 0), // g_11
149 barycentric_cubic_3d<4>(metric_array, i0, j0, k0, p.pos(0),
150 p.pos(1), p.pos(2), dx, p_lo,
151 1), // g_12 & g_21
152 barycentric_cubic_3d<4>(metric_array, i0, j0, k0, p.pos(0),
153 p.pos(1), p.pos(2), dx, p_lo,
154 2), // g_13 & g_31
155 barycentric_cubic_3d<4>(metric_array, i0, j0, k0, p.pos(0),
156 p.pos(1), p.pos(2), dx, p_lo,
157 3), // g_22
158 barycentric_cubic_3d<4>(metric_array, i0, j0, k0, p.pos(0),
159 p.pos(1), p.pos(2), dx, p_lo,
160 4), // g_23, g_32
161 barycentric_cubic_3d<4>(metric_array, i0, j0, k0, p.pos(0),
162 p.pos(1), p.pos(2), dx, p_lo, 5)}; // g_33
163
164 const CCTK_REAL inv_det_gamma =
165 1.0 / (gamma_x[0] * gamma_x[3] * gamma_x[5] +
166 2. * gamma_x[1] * gamma_x[2] * gamma_x[4] -
167 gamma_x[2] * gamma_x[2] * gamma_x[3] -
168 gamma_x[4] * gamma_x[4] * gamma_x[0] -
169 gamma_x[1] * gamma_x[1] * gamma_x[5]);
170
171 const amrex::GpuArray<CCTK_REAL, 6> gamma_inv_x = {
172 (gamma_x[3] * gamma_x[5] - gamma_x[4] * gamma_x[4]) * inv_det_gamma,
173 (gamma_x[4] * gamma_x[2] - gamma_x[1] * gamma_x[5]) * inv_det_gamma,
174 (gamma_x[1] * gamma_x[4] - gamma_x[2] * gamma_x[3]) * inv_det_gamma,
175 (gamma_x[0] * gamma_x[5] - gamma_x[2] * gamma_x[2]) * inv_det_gamma,
176 (gamma_x[2] * gamma_x[1] - gamma_x[0] * gamma_x[4]) * inv_det_gamma,
177 (gamma_x[0] * gamma_x[3] - gamma_x[1] * gamma_x[1]) *
178 inv_det_gamma};
179
180 // Compute a random initial velocity
181 ratio[1] = 2.0 * amrex::Random(engine) - 1.0;
182 ratio[2] = 2.0 * amrex::Random(engine) - 1.0;
183 ratio[0] = 2.0 * amrex::Random(engine) - 1.0;
184
185 // Normalizing the velocity.
186 const CCTK_REAL v_squared = ratio[0] * ratio[0] * gamma_inv_x[0] +
187 ratio[1] * ratio[1] * gamma_inv_x[3] +
188 ratio[2] * ratio[2] * gamma_inv_x[5] +
189 2.0 * ratio[0] * ratio[1] * gamma_inv_x[1] +
190 2.0 * ratio[0] * ratio[2] * gamma_inv_x[2] +
191 2.0 * ratio[1] * ratio[2] * gamma_inv_x[4];
192 const CCTK_REAL v = std::sqrt(v_squared);
193
194 // Create the particle and add it to the container
195 arrdata[StructType::vx][pidx] = ratio[0] / v;
196 arrdata[StructType::vy][pidx] = ratio[1] / v;
197 arrdata[StructType::vz][pidx] = ratio[2] / v;
198
199 // Update the particles counter
200 ++pidx;
201 }
202 });
203 CCTK_VINFO("%d particles created",
204 particle_tile.GetArrayOfStructs().size());
205 }

207} // random_photons_initializer

522 (gamma_x[4] * gamma_x[2] - gamma_x[1] * gamma_x[5]) * inv_det_gamma,
523 (gamma_x[1] * gamma_x[4] - gamma_x[2] * gamma_x[3]) * inv_det_gamma,
524 (gamma_x[0] * gamma_x[5] - gamma_x[2] * gamma_x[2]) * inv_det_gamma,
525 (gamma_x[2] * gamma_x[1] - gamma_x[0] * gamma_x[4]) * inv_det_gamma,
526 (gamma_x[0] * gamma_x[3] - gamma_x[1] * gamma_x[1]) * inv_det_gamma};
527
528 // Normalizing the velocity.
529 const CCTK_REAL v_squared = ratio[0] * ratio[0] * gamma_inv_x[0] +
530 ratio[1] * ratio[1] * gamma_inv_x[3] +
531 ratio[2] * ratio[2] * gamma_inv_x[5] +
532 2.0 * ratio[0] * ratio[1] * gamma_inv_x[1] +
533 2.0 * ratio[0] * ratio[2] * gamma_inv_x[2] +
534 2.0 * ratio[1] * ratio[2] * gamma_inv_x[4];
535
536 const CCTK_REAL v = std::sqrt(v_squared);
537
538 // Create the particle and add it to the container
539 arrdata[StructType::vx][i] = ratio[0] / v;
540 arrdata[StructType::vy][i] = ratio[1] / v;
541 arrdata[StructType::vz][i] = ratio[2] / v;
542 arrdata[StructType::ln_E][i] = 0;
543 }
544 }
