98template <typename StructType>
99void PhotonsContainer<StructType>::normalize_velocity(
100 const amrex::MultiFab &metric, const int level) {
101
102 // Get the with of the discretization on each direction.
103 const auto dx = this->Geom(level).CellSizeArray();
104 // Get the lower and higher value over the ParticleContainer Geometry
105 const auto p_lo = this->Geom(level).ProbLoArray();
106 const auto p_hi = this->Geom(level).ProbHiArray();
107
108 for (amrex::MFIter mfi = this->MakeMFIter(level); mfi.isValid(); ++mfi) {
109
110 // Get a reference to the particles
111 auto &particle_tile = this->DefineAndReturnParticleTile(level, mfi);
112
113 // Determines the current size and the required new size
114 const auto current_size = particle_tile.GetArrayOfStructs().size();
115
116 // Gets raw pointers to the two different ways particle data is stored for
117 // performance reasons: Array of Struct (AoS) and Struct of Arrays (SoA)
118 auto *p_struct = particle_tile.GetArrayOfStructs()().data();
119 auto arrdata = particle_tile.GetStructOfArrays().realarray();
120
121 // get the current process id
122 const auto metric_array = metric.array(mfi);
123
124 amrex::ParallelFor(current_size, [=] AMREX_GPU_DEVICE(int i) noexcept {
125 // Start a for loop with Random Number evolution for the velocity
126 const amrex::Real ratio[AMREX_SPACEDIM] = {arrdata[StructType::vx][i],
127 arrdata[StructType::vy][i],
128 arrdata[StructType::vz][i]};
129
130 // Generate a random position
131 const auto &p = p_struct[i];
132
133 const int i0 = amrex::Math::floor((p.pos(0) - p_lo[0]) / dx[0]);
134 const int j0 = amrex::Math::floor((p.pos(1) - p_lo[1]) / dx[1]);
135 const int k0 = amrex::Math::floor((p.pos(2) - p_lo[2]) / dx[2]);
136
137 // Interpolate metric
138 amrex::GpuArray<CCTK_REAL, 6> gamma_x;
139 interpolate_array<5>(gamma_x, metric_array, i0, j0, k0, p.pos(0),
140 p.pos(1), p.pos(2), dx, p_lo);
141
142 const CCTK_REAL inv_det_gamma =
143 1.0 / (gamma_x[0] * gamma_x[3] * gamma_x[5] +
144 2. * gamma_x[1] * gamma_x[2] * gamma_x[4] -
145 gamma_x[2] * gamma_x[2] * gamma_x[3] -
146 gamma_x[4] * gamma_x[4] * gamma_x[0] -
147 gamma_x[1] * gamma_x[1] * gamma_x[5]);
148
149 const amrex::GpuArray<CCTK_REAL, 6> gamma_inv_x = {
150 (gamma_x[3] * gamma_x[5] - gamma_x[4] * gamma_x[4]) * inv_det_gamma,
151 (gamma_x[4] * gamma_x[2] - gamma_x[1] * gamma_x[5]) * inv_det_gamma,
152 (gamma_x[1] * gamma_x[4] - gamma_x[2] * gamma_x[3]) * inv_det_gamma,
153 (gamma_x[0] * gamma_x[5] - gamma_x[2] * gamma_x[2]) * inv_det_gamma,
154 (gamma_x[2] * gamma_x[1] - gamma_x[0] * gamma_x[4]) * inv_det_gamma,
155 (gamma_x[0] * gamma_x[3] - gamma_x[1] * gamma_x[1]) * inv_det_gamma};
156
157 // Normalizing the velocity.
158 const CCTK_REAL v_squared = ratio[0] * ratio[0] * gamma_inv_x[0] +
159 ratio[1] * ratio[1] * gamma_inv_x[3] +
160 ratio[2] * ratio[2] * gamma_inv_x[5] +
161 2.0 * ratio[0] * ratio[1] * gamma_inv_x[1] +
162 2.0 * ratio[0] * ratio[2] * gamma_inv_x[2] +
163 2.0 * ratio[1] * ratio[2] * gamma_inv_x[4];
164
165 const CCTK_REAL v = std::sqrt(v_squared);
166
167 arrdata[StructType::vx][i] = ratio[0] / v;
168 arrdata[StructType::vy][i] = ratio[1] / v;
169 arrdata[StructType::vz][i] = ratio[2] / v;
170 });
171 }
172} // PhotonsContainer::normalize_velocity
